cmake_minimum_required(VERSION 3.16)
project(sm64ds_steve_core LANGUAGES CXX)
add_library(steve_core core.cpp)
target_compile_features(steve_core PUBLIC cxx_std_17)
target_include_directories(steve_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(steve_core_test core_test.cpp)
target_link_libraries(steve_core_test PRIVATE steve_core)
enable_testing()
add_test(NAME steve_core COMMAND steve_core_test)
